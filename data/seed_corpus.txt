def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def divide(a, b):
    if b == 0:
        return None
    return a / b

def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

def fibonacci(n):
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

def is_even(n):
    return n % 2 == 0

def is_odd(n):
    return n % 2 != 0

def max_of_two(a, b):
    if a > b:
        return a
    return b

def min_of_two(a, b):
    if a < b:
        return a
    return b

def clamp(value, low, high):
    if value < low:
        return low
    if value > high:
        return high
    return value

def absolute(n):
    if n < 0:
        return -n
    return n

def power(base, exp):
    result = 1
    for _ in range(exp):
        result *= base
    return result

def sum_list(nums):

def average(nums):
    if not nums:
        return 0
    return sum(nums) / len(nums)

def contains(items, target):

def find_index(items, target):
    for i, item in enumerate(items):
        if item == target:

def reverse_list(items):

    for i in range(len(items) - 1, -1, -1):

def unique(items):
    seen = []
    for item in items:

def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

def lcm(a, b):
    if a == 0 or b == 0:
        return 0
    return abs(a * b) // gcd(a, b)

def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True

def primes_until(limit):
    result = []
    for n in range(2, limit):
        if is_prime(n):
            result.append(n)
    return result

def bubble_sort(items):
    arr = list(items)
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

def linear_search(items, target):
    for i, item in enumerate(items):
        if item == target:
            return i
    return -1

def binary_search(items, target):
    low = 0
    high = len(items) - 1
    while low <= high:
        mid = (low + high) // 2
        if items[mid] == target:
            return mid
        if items[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return -1

def normalize_list(nums):

def median(nums):
    if not nums:
        return 0
    arr = sorted(nums)
    mid = len(arr) // 2
    if len(arr) % 2 == 0:
        return (arr[mid - 1] + arr[mid]) / 2
    return arr[mid]

def mode(nums):
    if not nums:
        return None
    counts = {}
    for n in nums:
        counts[n] = counts.get(n, 0) + 1
    best = nums[0]
    best_count = 0
    for key, value in counts.items():
        if value > best_count:
            best = key
            best_count = value
    return best

def pad_left(text, width, fill=" "):
    if len(text) >= width:
        return text
    return fill * (width - len(text)) + text

def pad_right(text, width, fill=" "):
    if len(text) >= width:
        return text
    return text + fill * (width - len(text))

def retry(fn, times):
    last_error = None
    for _ in range(times):
        try:
            return fn()
        except Exception as exc:
            last_error = exc
    raise last_error

def apply_all(fns, value):

    for fn in fns:

    return result

def memo_square(cache, n):
    if n in cache:
        return cache[n]
    cache[n] = n * n
    return cache[n]

def matrix_add(a, b):
    result = []
    for i in range(len(a)):
        row = []
        for j in range(len(a[i])):
            row.append(a[i][j] + b[i][j])
        result.append(row)
    return result

def matrix_shape(matrix):
    if not matrix:
        return 0, 0
    return len(matrix), len(matrix[0])

def transpose(matrix):
    if not matrix:
        return []
    rows = len(matrix)
    cols = len(matrix[0])
    result = []
    for j in range(cols):
        row = []
        for i in range(rows):
            row.append(matrix[i][j])
        result.append(row)
    return result
